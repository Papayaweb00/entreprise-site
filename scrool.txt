        // On détecte le défilement de la page
        $(window).scroll(function () {
            if ($(this).scrollTop() > 50) {  // Si l'on a défilé de plus de 50px
                $('#header').addClass('scrolled'); // Ajouter la classe 'scrolled'
            } else {
                $('#header').removeClass('scrolled'); // Retirer la classe 'scrolled'
